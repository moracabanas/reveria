- `TimesFM2p5Model` - from Google, 0.2B params, CPU-friendly

from darts.models import TimesFM2p5Model
from darts import TimeSeries
import pandas as pd
import numpy as np

# Create sample input
sample_data = pd.Series(np.random.randn(100).cumsum())
ts = TimeSeries.from_series(sample_data)

# Load model (CPU mode)
model = TimesFM2p5Model(device='cpu')
model.load()

# Predict
prediction = model.predict(n=10, inputs=ts)
print(f'PREDICTION_SHAPE: {len(prediction.values())}')
